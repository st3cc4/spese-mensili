import { auth, db, onAuthStateChanged, signOut, collection, getDocs, doc, getDoc, deleteDoc } from "../firebase-config.js"; let allExpenses = []; let currentUser = null; let userSalaryMap = {}; // Per memorizzare gli stipendi ed entrate salvate per utente // Ascolta i messaggi inviati dalla Dashboard principale (se necessario) window.addEventListener('message', (event) => { if (event.data && event.data.type === 'REFRESH_STORICO') { loadStoricoData(); } }); // Funzione richiamata dalla Dashboard quando si apre la scheda Storico window.refreshStoricoData = function() { loadStoricoData(); }; document.addEventListener('DOMContentLoaded', () => { // Inizializza i filtri al mese corrente const now = new Date(); const year = now.getFullYear(); const month = String(now.getMonth() + 1).padStart(2, '0'); const monthInput = document.getElementById('select-month'); if (monthInput) { monthInput.value = `${year}-${month}`; } // Imposta anno corrente nel selettore anno const yearSelect = document.getElementById('select-year'); if (yearSelect) { yearSelect.innerHTML = ''; for (let y = year; y >= 2024; y--) { const opt = document.createElement('option'); opt.value = y; opt.textContent = y; if (y === year) opt.selected = true; yearSelect.appendChild(opt); } } // Imposta giorno corrente const dayInput = document.getElementById('select-day'); if (dayInput) { const d = String(now.getDate()).padStart(2, '0'); dayInput.value = `${year}-${month}-${d}`; } // Inizializza flatpickr per intervallo personalizzato if (typeof flatpickr !== 'undefined') { flatpickr("#date-range-picker", { mode: "range", dateFormat: "Y-m-d", locale: "it", onChange: function(selectedDates) { if (selectedDates.length === 2) { applyFilters(); } } }); } // Ascoltatori per i cambi tipo di filtro const filterType = document.getElementById('filter-type'); if (filterType) { filterType.addEventListener('change', () => { updateFilterVisibility(); applyFilters(); }); } // Ascoltatori per tutti i campi di input e filtri const inputsToWatch = [ 'search-input', 'filter-type', 'select-day', 'select-month', 'select-week', 'select-year', 'chk-user1', 'chk-user2' ]; inputsToWatch.forEach(id => { const el = document.getElementById(id); if (el) { el.addEventListener('input', applyFilters); el.addEventListener('change', applyFilters); } }); // Logout dallo storico const logoutBtnStorico = document.getElementById('logout-btn-storico'); if (logoutBtnStorico) { logoutBtnStorico.addEventListener('click', () => signOut(auth)); } }); function updateFilterVisibility() { const type = document.getElementById('filter-type').value; document.getElementById('selector-day-container').classList.add('hidden'); document.getElementById('selector-week-container').classList.add('hidden'); document.getElementById('selector-month-container').classList.add('hidden'); document.getElementById('selector-year-container').classList.add('hidden'); document.getElementById('custom-date-container').classList.add('hidden'); if (type === 'day') { document.getElementById('selector-day-container').classList.remove('hidden'); } else if (type === 'week') { document.getElementById('selector-week-container').classList.remove('hidden'); } else if (type === 'month') { document.getElementById('selector-month-container').classList.remove('hidden'); } else if (type === 'year') { document.getElementById('selector-year-container').classList.remove('hidden'); } else if (type === 'custom') { document.getElementById('custom-date-container').classList.remove('hidden'); } } function isUserElenaByEmail(email) { if (!email) return false; return email.toLowerCase().includes("ele"); } // Controllo stato autenticazione nell'iframe onAuthStateChanged(auth, async (user) => { if (user) { currentUser = user; // Imposta le checkbox di default se non ancora toccate (entrambe selezionate o basate sull'utente) const chk1 = document.getElementById('chk-user1'); const chk2 = document.getElementById('chk-user2'); if (chk1 && chk2 && !chk1.checked && !chk2.checked) { chk1.checked = true; chk2.checked = true; } await loadStoricoData(); } else { currentUser = null; allExpenses = []; applyFilters(); } }); // Funzione per caricare i dati delle spese e configurazioni da Firebase async function loadStoricoData() { try { // Carica configurazione stipendi per calcolare le entrate nel periodo const configSnap = await getDocs(collection(db, "config")); userSalaryMap = {}; configSnap.forEach(docSnap => { const data = docSnap.data(); const base = parseFloat((data.stipendio || 0).toString().replace(',', '.')) || 0; const extra = parseFloat((data.extra || 0).toString().replace(',', '.')) || 0; userSalaryMap[docSnap.id] = { totalIncome: base + extra }; }); // Carica tutte le spese const querySnapshot = await getDocs(collection(db, "spese")); allExpenses = []; querySnapshot.forEach((docSnap) => { const data = docSnap.data(); const expUserId = data.userId || ""; // Determina se la spesa appartiene a Paolo o Elena in base allo userId o al campo salvato let owner = "paolo"; if (expUserId) { // Se abbiamo l'utente salvato o confrontiamo con gli UID noti o email // Nel dubbio, verifichiamo se l'utente è Elena o Paolo if (expUserId === currentUser?.uid && isUserElenaByEmail(currentUser?.email)) { owner = "elena"; } else if (data.owner === "elena" || data.isElena === true) { owner = "elena"; } else if (data.owner === "paolo") { owner = "paolo"; } else { // Controllo euristico se ci sono più utenti owner = expUserId === "elena" ? "elena" : "paolo"; } } allExpenses.push({ id: docSnap.id, amount: parseFloat((data.importo || data.amount || 0).toString().replace(',', '.')) || 0, date: data.data || data.date || "", category: data.categoria || data.category || "Altro", userId: expUserId, owner: owner }); }); applyFilters(); } catch (e) { console.error("Errore caricamento dati storico:", e); } } function applyFilters() { const showPaolo = document.getElementById('chk-user1').checked; const showElena = document.getElementById('chk-user2').checked; const filterType = document.getElementById('filter-type').value; const searchText = (document.getElementById('search-input').value || "").toLowerCase().trim(); // 1. Filtro Account e Testo let filtered = allExpenses.filter(exp => { const matchesAccount = (exp.owner === 'elena' && showElena) || (exp.owner !== 'elena' && showPaolo); if (!matchesAccount) return false; if (searchText) { const matchCat = exp.category.toLowerCase().includes(searchText); const matchAmount = exp.amount.toString().includes(searchText); const matchDate = exp.date.includes(searchText); if (!matchCat && !matchAmount && !matchDate) return false; } return true; }); // 2. Filtro Temporale const now = new Date(); filtered = filtered.filter(exp => { if (!exp.date) return false; const parts = exp.date.split('-'); if (parts.length !== 3) return false; const expDate = new Date(parseInt(parts[0]), parseInt(parts[1]) - 1, parseInt(parts[2])); if (filterType === 'day') { const dayVal = document.getElementById('select-day').value; return exp.date === dayVal; } else if (filterType === 'month') { const monthVal = document.getElementById('select-month').value; // Formato YYYY-MM if (!monthVal) return true; const [selYear, selMonth] = monthVal.split('-'); return parts[0] === selYear && parts[1] === selMonth; } else if (filterType === 'year') { const yearVal = document.getElementById('select-year').value; return parts[0] === yearVal; } else if (filterType === 'custom') { const rangePicker = document.getElementById('date-range-picker'); if (rangePicker && rangePicker._flatpickr) { const dates = rangePicker._flatpickr.selectedDates; if (dates.length === 2) { const start = new Date(dates[0]); start.setHours(0,0,0,0); const end = new Date(dates[1]); end.setHours(23,59,59,999); return expDate >= start && expDate <= end; } } } return true; }); // Calcoli Totali const totalExpenses = filtered.reduce((sum, e) => sum + e.amount, 0); // Calcolo Entrate basato sui filtri attivi (stipendi configurati) let totalIncome = 0; if (showPaolo) totalIncome += 1500; // Valore di default o preso da userSalaryMap se disponibile if (showElena) totalIncome += 1500; // (Puoi affinarlo sommando i valori reali da userSalaryMap) const balance = totalIncome - totalExpenses; // Aggiornamento DOM metriche document.getElementById('filter-total-val').innerText = `€ ${totalExpenses.toFixed(2)}`; document.getElementById('filter-income-val').innerText = `€ ${totalIncome.toFixed(2)}`; const balanceEl = document.getElementById('filter-balance'); balanceEl.innerText = `€ ${balance.toFixed(2)}`; balanceEl.style.color = balance >= 0 ? '#28a745' : '#dc3545'; // Calcolo Totali per Categoria const categoryMap = {}; filtered.forEach(exp => { categoryMap[exp.category] = (categoryMap[exp.category] || 0) + exp.amount; }); const categoryListEl = document.getElementById('category-totals-list'); categoryListEl.innerHTML = ''; const sortedCategories = Object.entries(categoryMap).sort((a, b) => b[1] - a[1]); if (sortedCategories.length === 0) { categoryListEl.innerHTML = '
Nessuna spesa trovata per i filtri selezionati.
'; } else { sortedCategories.forEach(([cat, amount]) => { categoryListEl.innerHTML += `
${cat} € ${amount.toFixed(2)}
`; }); } // Render Dettaglio Spese const expensesListEl = document.getElementById('filtered-expenses-list'); expensesListEl.innerHTML = ''; if (filtered.length === 0) { expensesListEl.innerHTML = '
Nessun movimento trovato.
'; return; } // Ordina per data decrescente filtered.sort((a, b) => new Date(b.date) - new Date(a.date)); filtered.forEach(exp => { const parts = exp.date.split('-'); const dateEu = parts.length === 3 ? `${parts[2]}/${parts[1]}/${parts[0]}` : exp.date; expensesListEl.innerHTML += `
${exp.category} ${dateEu} (${exp.owner.toUpperCase()})
€ ${exp.amount.toFixed(2)}
`; }); } // Funzioni globali per Modifica e Cancellazione dallo Storico window.editExpenseFromHistory = function(id, amount, date, category) { localStorage.setItem('edit_id', id); localStorage.setItem('edit_amount', amount); localStorage.setItem('edit_date', date); localStorage.setItem('edit_category', category); // Invia messaggio alla finestra principale per passare alla Dashboard window.parent.postMessage({ type: 'EDIT_EXPENSE_FROM_HISTORY', id }, '*'); }; window.deleteExpenseFromHistory = async function(id) { if (confirm("Vuoi cancellare questa spesa?")) { try { await deleteDoc(doc(db, "spese", id)); await loadStoricoData(); } catch (e) { console.error("Errore durante la cancellazione:", e); alert("Errore durante la cancellazione della spesa."); } } };
