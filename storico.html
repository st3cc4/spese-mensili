function renderResults(list, from, to, type) { const totalSpent = list.reduce((sum, e) => sum + e.amount, 0); document.getElementById('filter-total-val').innerText = totalSpent.toFixed(2); let totalIncome = 0; const monthlyIncome = (salaryConfig.base || 0) + (salaryConfig.extra || 0); // Logica delle entrate basata sui tuoi cicli di stipendio if (type === 'month' || type === 'day' || type === 'week') { // Per Mese, Giorno o Settimana, l'entrata di riferimento è l'ultimo stipendio percepito con i suoi extra totalIncome = monthlyIncome; } else if (type === 'year') { // Per l'anno, sommiamo i cicli di stipendio (12 mensilità standard con i rispettivi extra) totalIncome = monthlyIncome * 12; } else if (type === 'custom' && from && to) { // Per un periodo personalizzato, proporzioniamo i giorni sul ciclo mensile const diffTime = Math.abs(to - from); const diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24)) + 1; totalIncome = (monthlyIncome / 30) * diffDays; } else { totalIncome = monthlyIncome; } document.getElementById('filter-income-val').innerText = totalIncome.toFixed(2); const netBalance = totalIncome - totalSpent; const balanceElement = document.getElementById('filter-balance'); balanceElement.innerText = `€ ${netBalance.toFixed(2)}`; balanceElement.style.color = netBalance >= 0 ? '#28a745' : '#dc3545'; // ... (il resto della funzione per le categorie e le spese rimane invariato)
